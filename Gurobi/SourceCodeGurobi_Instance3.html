import pandas as pd
import numpy as np
from math import sqrt
import gurobipy as gp
from gurobipy import GRB
import time
import matplotlib.pyplot as plt
import matplotlib.patches as mpatches

start_time = time.time()

# STEP 1: Load instance_3 files
cities = pd.read_csv(r"C:\Users\User\Desktop\Code\Cities_3.txt", sep=r'\s+', engine='python')
depots = pd.read_csv(r"C:\Users\User\Desktop\Code\Depots_3.txt", sep=r'\s+', engine='python')

sales_data = []
with open(r"C:\Users\User\Desktop\Code\Salespersons_3.txt", 'r') as f:
    next(f)
    for line in f:
        parts = line.strip().split()
        if len(parts) == 6:
            sid, x, y, color, vel, depot = parts
            color_list = [int(color)]
        elif len(parts) == 7:
            sid, x, y, color1, color2, vel, depot = parts
            color_list = [int(color1), int(color2)]
        else:
            continue
        sales_data.append({
            'Salesperson': int(sid),
            'X': float(x),
            'Y': float(y),
            'Color': color_list,
            'Velocity': float(vel),
            'sourceDepot': int(depot)
        })

sales = pd.DataFrame(sales_data)

# STEP 2: Index sets and base data
V = cities.index.tolist()
S = sales.index.tolist()
color_cities = cities['Color'].tolist()
color_sales = sales['Color'].tolist()
dur = {(i, s): cities.loc[i, 'Duration'] for i in V for s in S}

# STEP 3: Coordinates
city_coords = cities[['X', 'Y']].to_dict(orient='index')
dept_coords = depots[['X', 'Y']].to_dict(orient='index')

node_coords = {s: city_coords.copy() for s in S}
src_nearest_city = {}
dst_nearest_city = {}
for s in S:
    node_coords[s]['src'] = {'X': dx, 'Y': dy}
    depot_index = int(sales.loc[s, 'sourceDepot']) % len(dept_coords)
    dx, dy = dept_coords[depot_index]['X'], dept_coords[depot_index]['Y']
    node_coords[s]['dst'] = {'X': dx, 'Y': dy}
    src_nearest_city[s] = min(city_coords.items(), key=lambda kv: sqrt((kv[1]['X'] - sx)**2 + (kv[1]['Y'] - sy)**2))[0]
    dst_nearest_city[s] = min(city_coords.items(), key=lambda kv: sqrt((kv[1]['X'] - dx)**2 + (kv[1]['Y'] - dy)**2))[0]

# STEP 4: Travel time omega(i,j,s)
omega = {}
for s in S:
    for i in list(city_coords) + ['src', 'dst']:
        for j in list(city_coords) + ['src', 'dst']:
            if i == j:
                omega[i, j, s] = 0
            else:
                xi, yi = node_coords[s][i]['X'], node_coords[s][i]['Y']
                xj, yj = node_coords[s][j]['X'], node_coords[s][j]['Y']
                omega[i, j, s] = sqrt((xi - xj)**2 + (yi - yj)**2) / sales.loc[s, 'Velocity']

# STEP 5: Precedence matrix
pi = np.zeros((len(V), len(V)), dtype=int)
for idx, row in cities.iterrows():
    if row['Precede'] != -1:
        pi[int(row['Precede']), idx] = 1

# STEP 6: Compatibility a(i,j,s)
a = {}
for s in S:
    for i in list(city_coords) + ['src', 'dst']:
        for j in list(city_coords) + ['src', 'dst']:
            if i == j:
                a[i, j, s] = 0
            elif i in V and j in V:
                a[i, j, s] = int(color_cities[i] in color_sales[s] and color_cities[j] in color_sales[s])
            elif i in V:
                a[i, j, s] = int(color_cities[i] in color_sales[s])
            elif j in V:
                a[i, j, s] = int(color_cities[j] in color_sales[s])
            else:
                a[i, j, s] = 1

# STEP 7: Build Gurobi model
model = gp.Model("ECTSP_Instance")

nodes = V + ['src', 'dst']
x = model.addVars(nodes, nodes, S, vtype=GRB.BINARY, name="x")
y = model.addVars(V, V, vtype=GRB.BINARY, name="y")
z = model.addVars(V, V, S, vtype=GRB.BINARY, name="z")
Q = model.addVar(vtype=GRB.CONTINUOUS, name="Q")
u = model.addVars(V, S, vtype=GRB.CONTINUOUS, name="u")

# STEP 8: Constraints
for s in S:
    for i in nodes:
        for j in nodes:
            if i != j:
                model.addConstr(x[i, j, s] <= a[i, j, s])

for j in V:
    model.addConstr(gp.quicksum(x[i, j, s] for s in S for i in nodes if i != j) == 1)

for i in V:
    model.addConstr(gp.quicksum(x[i, j, s] for s in S for j in nodes if i != j) == 1)

for s in S:
    model.addConstr(gp.quicksum(x['src', j, s] for j in V) == 1)
    model.addConstr(gp.quicksum(x[i, 'dst', s] for i in V) == 1)

model.addConstr(gp.quicksum(x['src', j, s] for s in S for j in V) >= 1)

for s in S:
    model.addConstr(gp.quicksum(x[i, 'src', s] for i in nodes if i != 'src') == 0)
    model.addConstr(gp.quicksum(x['dst', j, s] for j in nodes if j != 'dst') == 0)

for s in S:
    for j in V:
        model.addConstr(gp.quicksum(x[i, j, s] for i in nodes if i != j) ==
                        gp.quicksum(x[j, k, s] for k in nodes if k != j))

for s in S:
    for i in nodes:
        model.addConstr(x[i, i, s] == 0)

for s in S:
    model.addConstr(gp.quicksum((omega[i, j, s] + (dur[i, s] if i in V else 0)) * x[i, j, s]
                                for i in nodes for j in nodes if i != j) <= Q)

for i in V:
    for j in V:
        for s in S:
            model.addConstr(y[i, j] >= x[i, j, s])

for i in V:
    for j in V:
        if i != j:
            model.addConstr(y[i, j] + y[j, i] == 1)

for i in V:
    for j in V:
        for k in V:
            if len({i, j, k}) == 3:
                model.addConstr(y[i, k] + y[k, j] + y[j, i] <= 2)

for s in S:
    for i in V:
        for j in V:
            if i != j and pi[i][j] == 1:
                if color_cities[i] in color_sales[s] and color_cities[j] in color_sales[s]:
                    model.addConstr(2 * z[i, j, s] ==
                        gp.quicksum(x[i, k, s] for k in nodes if k != i) +
                        gp.quicksum(x[k, j, s] for k in nodes if k != j))

for i in V:
    for j in V:
        if i != j and pi[i][j] == 1:
            model.addConstr(y[i, j] == gp.quicksum(z[i, j, s] for s in S
                                                   if color_cities[i] in color_sales[s] and color_cities[j] in color_sales[s]))


model.setObjective(
    gp.quicksum((omega[i, j, s] + (dur[i, s] if i in V else 0)) * x[i, j, s]
                for i in nodes for j in nodes if i != j for s in S), GRB.MINIMIZE)

model.setParam("TimeLimit", 600)
model.optimize()
