import pandas as pd
import matplotlib.pyplot as plt

# === Step 1: Load data ===
cities = pd.read_csv("C:/Users/User/Desktop/Project1/Cities_0.txt", sep=r"\s+", engine="python")
depots = pd.read_csv("C:/Users/User/Desktop/Project1/Depots_0.txt", sep=r"\s+", engine="python")

sales_data = []
with open("C:/Users/User/Desktop/Project1/Salespersons_0.txt", 'r') as f:
    next(f)
    for line in f:
        parts = line.strip().split()
        sid = int(parts[0])
        x, y = float(parts[1]), float(parts[2])
        color_split_idx = 3 if parts[4].replace('.', '', 1).isdigit() else 4
        colors = list(map(int, parts[3:color_split_idx]))
        velocity = float(parts[color_split_idx])
        depot_id = int(parts[color_split_idx + 1])
        sales_data.append([sid, x, y, colors, velocity, depot_id])
salespersons = pd.DataFrame(sales_data, columns=["Salesperson", "X", "Y", "Color", "Velocity", "sourceDepot"])

# === Step 2: Load chromosome ===
def load_chromosome(filepath):
    with open(filepath, 'r') as f:
        lines = f.readlines()
    start = lines.index('--- FINAL_CHROMOSOME_START ---\n') + 1
    end = lines.index('--- FINAL_CHROMOSOME_END ---\n')
    chromo = {}
    for line in lines[start:end]:
        sp, route = line.strip().split(":")
        chromo[int(sp)] = route.strip().split()
    return chromo

# === Step 3: Plot ===
def plot_salesperson_routes(cities_df, depots_df, chromosome, salespersons_df, title="Final Tour"):
    fig, ax = plt.subplots(figsize=(12, 10))

    for _, row in cities_df.iterrows():
        ax.plot(row['X'], row['Y'], 'o', color='dodgerblue', markersize=10)
        ax.text(row['X'], row['Y'] + 2500, f"City {int(row['City'])}", fontsize=6, ha='center')

    for i, row in depots_df.iterrows():
        ax.plot(row['X'], row['Y'], 's', color='gold', markersize=12)
        ax.text(row['X'], row['Y'] + 3000, f"Depot {i}", fontsize=9, ha='center')

    colors = ['crimson', 'mediumblue', 'darkorange', 'seagreen', 'purple', 'brown', 'teal', 'magenta']
    color_cycle = iter(colors)

    offset_map = {
        0: (-10000, 14000),
        1: (12000, 12000),
        2: (-8000, -8000),
        3: (-14000, -10000),
        4: (0, 8000),
        5: (0, -8000),
        6: (8000, 0),
        7: (-8000, 0),
    }

    for sp, route in chromosome.items():
        if not route:
            continue

        color = next(color_cycle, 'gray')
        depot_idx = int(salespersons_df.loc[salespersons_df["Salesperson"] == sp, "sourceDepot"].iloc[0])
        depot = depots_df.iloc[depot_idx % len(depots_df)]
        depot_x, depot_y = depot['X'], depot['Y']

        tour_ids = [int(cid) for cid in route]
        coords = [(depot_x, depot_y)] + [
            cities_df.loc[cities_df['City'] == cid, ['X', 'Y']].values[0] for cid in tour_ids
        ] + [(depot_x, depot_y)]

        for i in range(len(coords) - 1):
            x_start, y_start = coords[i]
            x_end, y_end = coords[i + 1]
            ax.annotate("",
                        xy=(x_end, y_end),
                        xytext=(x_start, y_start),
                        arrowprops=dict(arrowstyle="->", color=color, lw=2))

        for idx, (x, y) in enumerate(coords[1:-1]):
            ax.text(x, y - 1000, f"{idx+1}", fontsize=7, color='black', ha='center')

        offset_start = offset_map.get(sp % len(offset_map), (0, 0))
        offset_end = (offset_start[0], offset_start[1] - 4000)

        x_start, y_start = coords[0]
        x_end, y_end = coords[-1]

        ax.text(x_start + offset_start[0], y_start + offset_start[1],
                f"SP {sp} Start", fontsize=9, color=color, ha='center',
                bbox=dict(facecolor='white', edgecolor=color, boxstyle='round,pad=0.3'))

        ax.text(x_end + offset_end[0], y_end + offset_end[1],
                f"SP {sp} End", fontsize=9, color=color, ha='center',
                bbox=dict(facecolor='white', edgecolor=color, boxstyle='round,pad=0.3'))

        ax.plot([], [], color=color, label=f"Salesperson {sp}")

    ax.set_title(title)
    ax.set_xlabel("X Coordinate")
    ax.set_ylabel("Y Coordinate")
    ax.grid(True)
    ax.legend(loc='upper left')
    plt.tight_layout()
    plt.show()

# === Step 4: Run plot ===
chromosome = load_chromosome("C:/Users/User/Desktop/Project1/best_chromosome_Instance0.txt")
plot_salesperson_routes(
    cities_df=cities,
    depots_df=depots,
    chromosome=chromosome,
    salespersons_df=salespersons,
    title="Best Chromosome Route"
)
